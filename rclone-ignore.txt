# ==============================================================================
# Rclone Ignore & Exclusion Rules
# Standard development build artifacts, dependencies, caches, and secrets.
# Syntax: '- ' to exclude from sync, '+ ' to include/preserve.
# In rclone, unanchored patterns (without leading '/') match both at root
# and at all nested directory depths.
# ==============================================================================

# Node / JavaScript / TypeScript
- node_modules/**
- .npm/**
- .yarn/**
- .pnpm-store/**

# Python
- __pycache__/**
- *.py[cod]
- .venv/**
- venv/**
- env/**
- .pytest_cache/**
- .mypy_cache/**
- .ruff_cache/**

# Rust
- target/**

# Mobile & Cross-Platform (Flutter / Android / iOS / Kotlin)
- .dart_tool/**
- .gradle/**
- build/**
- Pods/**
- .symlinks/**
- DerivedData/**
- *.apk
- *.aab
- *.ipa

# Web / Framework build outputs & caches
- dist/**
- out/**
- .next/**
- .nuxt/**
- .svelte-kit/**
- .turbo/**

# Version control repositories & internal object caches
# (Prevents syncing loose git hashes that exhaust cloud API TPS and trigger rate-limits)
- .git/**

# IDEs, OS metadata & system cache files
- .idea/**
- .vscode/**
- .vs/**
- .DS_Store
- .DS_Store?
- ._*
- Thumbs.db
- desktop.ini

# Environment variables & secrets (preserve .env.example)
+ .env.example
- .env
- .env.*
- *.key
- *.jks
- *.keystore

# OS duplicate and backup files & folders across all directories
# Matches: 'folder - copia', 'folder - copia (2)', 'folder - Copy', 'folder - Copy (2)', 'folder copy'
- * - copia*/**
- * - copia*
- * - Copy*/**
- * - Copy*
- * copy*/**
- * copy*

# ==============================================================================
# Semantic Policy Directives:
# Standard exclusions ('- ') are treated as transient bloat (excluded from sync,
# and purged on remote during 'cleanup').
#
# To permanently protect local private archives, offline keys, or datasets so
# they are NEVER synced, but also NEVER deleted from cloud storage during 'cleanup',
# precede the rule with '# @local_only' or use inline prefix '-! ':
#
# Examples:
#   # @local_only
#   - local_offline_data/**
#   -! private_keys/**
# ==============================================================================
